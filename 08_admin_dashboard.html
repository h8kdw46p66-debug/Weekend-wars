CREATE OR REPLACE FUNCTION public.get_weekend_completed_missions() RETURNS TABLE ( submission_id uuid, player_id uuid, player_name text, mission_title text ) LANGUAGE sql STABLE SECURITY DEFINER SET search_path TO '' AS $$ SELECT s.id AS submission_id, p.id AS player_id, p.display_name AS player_name, m.title AS mission_title FROM public.submissions s JOIN public.mission_assignments ma ON ma.id = s.assignment_id JOIN public.players p ON p.id = ma.player_id JOIN public.missions m ON m.id = ma.mission_id WHERE s.status = 'approved' AND p.weekend_id = m.weekend_id AND p.weekend_id IN ( SELECT admin.weekend_id FROM public.players admin WHERE admin.auth_user_id = auth.uid() AND admin.role = 'admin' ) ORDER BY p.display_name, m.title; $$; REVOKE ALL ON FUNCTION public.get_weekend_completed_missions() FROM PUBLIC; GRANT EXECUTE ON FUNCTION public.get_weekend_completed_missions() TO authenticated;
